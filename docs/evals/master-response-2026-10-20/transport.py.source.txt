"""Exercise the unchanged asynchronous client via loopback; never accesses paid API."""
import contextlib
import importlib.util
import io
import json
import os
import threading
import time
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

os.environ['QT_QPA_PLATFORM']='offscreen'
from PySide6.QtWidgets import QApplication
from rpg_vkr.narrative.ollama import OllamaClient
from rpg_vkr.narrative.context import build_frame
from rpg_vkr.narrative.master import campaign_context
from rpg_vkr.narrative.dialogue import last_interlocutor
from rpg_vkr.storage.knowledge import retrieve
spec=importlib.util.spec_from_file_location('independent_audit',Path(__file__).with_name('audit.py'))
audit=importlib.util.module_from_spec(spec)
with contextlib.redirect_stdout(io.StringIO()): spec.loader.exec_module(audit)
app=QApplication.instance() or QApplication([])
results=[]
selected=['ordinary_observation','brief_yes','received_answer','wellbeing_synonym','new_npc_name_inflected',
          'new_npc_name_direct','unsupported_magic_obstacle','exploration','partial_failure_prefix','committed_attack_literal',
          'fake_reward_indirect']
work=[(id,next(c for c in audit.cases if c['id']==id),'normal') for id in selected]
base=next(c for c in audit.cases if c['id']=='exploration')
work.extend((kind,base,kind) for kind in ('http_503','length','invalid_json'))
for id,c,mode in work:
    calls=[]
    class Handler(BaseHTTPRequestHandler):
        def log_message(self,*args):pass
        def do_POST(self):
            body=json.loads(self.rfile.read(int(self.headers.get('Content-Length','0'))))
            calls.append({'model':body['model'],'repair_requested':'Первая попытка отклонена' in body['messages'][0]['content'],
                          'limit':body.get('options',{}).get('num_predict')})
            status=503 if mode=='http_503' else 200
            raw=b'{broken' if mode=='invalid_json' else json.dumps({
                'message':{'role':'assistant','content':c['text']},'done':True,
                'done_reason':'length' if mode=='length' else 'stop'},ensure_ascii=False).encode()
            self.send_response(status);self.send_header('Content-Length',str(len(raw)));self.end_headers();self.wfile.write(raw)
    server=ThreadingHTTPServer(('127.0.0.1',0),Handler)
    thread=threading.Thread(target=server.serve_forever,daemon=True);thread.start()
    client=OllamaClient(base_url=f'http://127.0.0.1:{server.server_port}')
    client.models=(audit.MODEL,)
    emissions=[];diagnostics=[];proposals=[]
    client.failed.connect(lambda text:emissions.append(('failed',text)))
    client.proposal_ready.connect(lambda value:(proposals.append(value),emissions.append(('proposal','accepted'))))
    client.narration_ready.connect(lambda value:emissions.append(('narration','accepted')))
    client.diagnostics_ready.connect(diagnostics.append)
    session=c['session'];before=session.to_dict(),session.dice.checkpoint()
    if c['confirmed']:
        client.narrate(build_frame(session),audit.MODEL,player_message=c['message'],plain=True,
                       dialogue=retrieve(session,c['message']),campaign=campaign_context(session),history=c['history'])
    else:
        client.develop(build_frame(session),audit.MODEL,c['message'],memory=retrieve(session,c['message']),
                       campaign=campaign_context(session,last_interlocutor(session)),history=c['history'])
    deadline=time.monotonic()+5
    while not emissions and time.monotonic()<deadline:app.processEvents();time.sleep(.003)
    assert emissions, id
    assert before==(session.to_dict(),session.dice.checkpoint()),id
    reasons=[d.split('Проверка: ',1)[1].split('\n',1)[0] for d in diagnostics]
    results.append(dict(id=id,mode=mode,expected_valid=c['valid'] if mode=='normal' else None,
                        response_count=len(calls),calls=calls,validation_reasons=reasons,
                        final=emissions[-1],state_and_rng_unchanged=True,
                        proposed_focus=proposals[-1].focus_id if proposals else None))
    client.cancel();client.deleteLater();app.processEvents();server.shutdown();server.server_close();thread.join()
summary={'cases':len(results),'model_inferences':0,'actual_external_requests':0,
         'loopback_post_responses':sum(r['response_count'] for r in results),
         'accepted_cases':sum(r['final'][0] in ('proposal','narration') for r in results),
         'failed_cases':sum(r['final'][0]=='failed' for r in results),
         'rejected_response_validations':sum(len(r['validation_reasons']) for r in results if r['final'][0]=='failed' and r['mode']!='http_503'),
         'network_failure_cases':sum(r['mode']=='http_503' for r in results),
         'after':None}
Path(__file__).with_name('transport.json').write_text(json.dumps({'summary':summary,'cases':results},ensure_ascii=False,indent=2),encoding='utf-8')
print(json.dumps(summary,ensure_ascii=False,indent=2))
for r in results:print(r['id'],r['response_count'],r['final'],r['validation_reasons'])
