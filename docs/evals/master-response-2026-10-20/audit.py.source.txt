def context_for(session, message, confirmed=False, history=()):
    return story_context(build_frame(session), MODEL, message,
        memory=retrieve(session, message), campaign=campaign_context(session, last_interlocutor(session)),
        confirmed=confirmed, history=history)

def response(text):
    return {'message':{'role':'assistant','content':text}, 'done':True, 'done_reason':'stop'}

results=[]
for c in cases:
    session=c['session']; before=(deepcopy(session.to_dict()),session.dice.checkpoint())
    context=context_for(session,c['message'],c['confirmed'],c['history'])
    accepted=False; error=''; proposal=None
    try:
        if c['confirmed']:
            text=story_text(response(c['text']),confirmed=True)
            check_story_quality(text,build_frame(session),c['message'],brief=context['turn_brief'],previous=c['history'],
                campaign={k:context.get(k) for k in ('hero_description','resolution','interaction_result','death_confirmed')})
        else:
            proposal=story_proposal(response(c['text']),build_frame(session),c['message'],context=context,previous=c['history'])
        accepted=True
    except (ValueError,TypeError,KeyError) as e:error=type(e).__name__+': '+str(e)
    assert before==(session.to_dict(),session.dice.checkpoint()), c['id']
    results.append({k:v for k,v in c.items() if k not in ('session','history')}|{
        'accepted':accepted,'error':error,'brief':context['turn_brief'],
        'focus_id':proposal.focus_id if proposal else None,
        'state_and_rng_unchanged':True})
