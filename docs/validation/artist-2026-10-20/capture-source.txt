from pathlib import Path
from tempfile import TemporaryDirectory
import hashlib,json
import pytest
from PySide6.QtWidgets import QApplication
from rpg_vkr.ui.main_window import MainWindow
from rpg_vkr.storage.slots import SaveSlots
from rpg_vkr.visuals.policy import FLUX
from test_artist_panel import image_catalog, local_transport, finish
from test_connection_metadata import metadata_server

app=QApplication([])
out=Path('docs/validation/artist-2026-10-20/screenshots');out.mkdir(exist_ok=True)
with TemporaryDirectory() as folder, pytest.MonkeyPatch.context() as patch:
    patch.delenv('RPG_VKR_ROUTERAI_API_KEY',raising=False)
    patch.delenv('RPG_VKR_LLM_API_KEY',raising=False)
    w=MainWindow(saves=SaveSlots(Path(folder)/'saves'))
    w.resize(1280,950);w.show();w.open_section('Мастер');app.processEvents()
    def snap(name):
        app.processEvents();w.grab().save(str(out/(name+'.png')))
    snap('master-1280')
    w.open_section('Художник')
    p=w.artist_panel
    p.shared_key.setChecked(False);p.key_input.setText('artist-synthetic-key')
    with metadata_server(patch,catalog=image_catalog()) as records:
        local_transport(p)
        p.check_button.click();finish(p,app)
        assert records==[('/key',True),('/models',False)]
        assert not w.visuals.registry.ledger()
        snap('artist-1280')
    p.model_choice.showPopup();app.processEvents()
    p.model_choice.view().grab().save(str(out/'models-popup.png'))
    p.model_choice.hidePopup()
    p.model_choice.setCurrentIndex(p.model_choice.findData(FLUX));p.model_choice.activated.emit(p.model_choice.currentIndex())
    snap('flux-details-1280')
    w.resize(640,620);app.processEvents();snap('artist-compact-top-640')
    scroll=w.tabs.currentWidget();scroll.verticalScrollBar().setValue(scroll.verticalScrollBar().maximum())
    snap('artist-compact-models-640')
    w.resize(1280,950);w.theme_choice.setCurrentText('Светлая')
    p.show_budget();app.processEvents();scroll.verticalScrollBar().setValue(scroll.verticalScrollBar().maximum())
    snap('artist-budget-light-1280')
    w.close()
manifest={'synthetic_loopback':True,'paid_requests':0,'files':[{'path':f.name,'sha256':hashlib.sha256(f.read_bytes()).hexdigest()} for f in sorted(out.glob('*.png'))]}
(out.parent/'screenshots-manifest.json').write_text(json.dumps(manifest,indent=2),encoding='utf-8')
print('7 Qt screenshots; synthetic key/catalog; only GET key+models; no paid request.')
